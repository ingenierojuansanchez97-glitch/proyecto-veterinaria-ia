======== Reglas clínicas + predicción combinada ======== def classify_band(vals: dict, especie: str = "Canino"): refs = REF_RANGES.get(especie, REF_RANGES["Canino"]) markers = ["Creatinina", "Urea", "BUN", "SDMA"] mild_count = 0 severe_count = 0 for m in markers: v = float(vals[m]) r = v / float(refs[m]) if r > 1.3: severe_count += 1 elif r > 1.0: mild_count += 1 if severe_count >= 1 or (mild_count + severe_count) >= 2: band_idx = 2 # Alto elif mild_count == 1 and severe_count == 0: band_idx = 1 # Medio else: band_idx = 0 # Bajo probs = np.zeros(3, dtype=float) probs[band_idx] = 0.80 return band_idx, probs def _pad3(arr): arr = np.array(arr, dtype=float) if arr.shape[0] == 3: return arr if arr.shape[0] == 2: arr = np.array([arr[0], 0.0, arr[1]]) s = arr.sum() return arr / s if s > 0 else np.array([1 / 3, 1 / 3, 1 / 3]) arr = arr[:3] s = arr.sum() return arr / s if s > 0 else np.array([1 / 3, 1 / 3, 1 / 3]) MODEL_BUNDLE = safe_load_model(MODEL_PATH) # -> (model, feat_order) def predict_blended(vals: dict, especie: str, model_bundle): # Reglas clínicas _, probs_rules = classify_band(vals, especie) # Modelo ML model, feat_order = model_bundle X = pd.DataFrame([vals])[feat_order] probs_ml = safe_predict_proba(model, X)[0] probs_ml = _pad3(probs_ml) # Mezcla final_probs = 0.30 * np.array(probs_rules) + 0.60 * np.array(probs_ml) final_probs = final_probs / final_probs.sum() pred_idx = int(np.argmax(final_probs)) return pred_idx, final_probs # ================== Helpers de sesión ================== def require_user(request: Request) -> Dict[str, Any]: user = request.session.get("user") if not user: #
